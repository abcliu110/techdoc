# 数学学习软件项目架构规范

> 版本：v1.0
> 创建日期：2026-10-06
> 状态：正式版

---

## 一、设计原则与核心目标

### 1.1 核心原则

| 原则 | 说明 | 约束 |
|------|------|------|
| 领域层纯 TypeScript | 领域模型、实体、值对象、领域服务不依赖任何框架或运行时 | 禁止在 `domain/` 目录内 import React/Zustand/SQL/Dexie |
| Repository 契约在上层 | Repository 接口定义在 `application/ports/`，实现在 `platform/` | 接口不能依赖具体存储技术 |
| 组件只负责渲染 | React 组件只做 UI 渲染，逻辑放入 hooks 或 store | 组件内禁止写业务判断逻辑 |
| 内容与代码分离 | 学习内容以 JSON 格式存储在 `content/` 目录 | 内容修改不触发代码构建 |

### 1.2 目标范围（第一阶段）

- **数学思想**：`数形结合`
- **学习内容**：分数大小比较
- **学习闭环**：情境 -> 操作 -> 表示 -> 猜想 -> 验证 -> 迁移

## 二、技术选型

| 层级 | 技术选型 | 说明 |
|------|---------|------|
| 跨端壳 | Tauri 2.x | 桌面端 Windows/macOS + Android 共用代码库 |
| 前端框架 | React 18 + TypeScript 5.x | 组件化开发 |
| 构建工具 | Vite 5.x | 开发体验好，热更新快 |
| 状态管理 | Zustand 4.x | 轻量级，TypeScript 友好 |
| 路由 | React Router 6.x | SPA 路由管理 |
| 持久化（桌面/Android） | SQLite（tauri-plugin-sql） | 本地优先，数据可靠 |
| 持久化（Web） | IndexedDB（Dexie） | 浏览器原生存储 |
| 内容校验 | Zod | JSON Schema 校验 |
| 数学判定 | 自研等价式判定器 | MVP 阶段不引入 CAS 库 |

### 4.2 领域服务

```typescript
// src/domain/services/LearningSessionService.ts

import {
  LearningSession,
  SessionStatus,
  LearningEvidence,
  SessionSnapshot,
} from '../types/learning-session';
import { DomainError } from '../errors';

/**
 * 学习会话领域服务
 */
export class LearningSessionService {
  /**
   * 创建新的学习会话
   */
  createSession(
    id: string,
    knowledgePointId: string,
    ability: Ability
  ): LearningSession {
    return {
      id,
      knowledgePointId,
      ability,
      status: SessionStatus.NOT_STARTED,
      currentStepIndex: 0,
      startedAt: Date.now(),
      evidence: [],
    };
  }

  /**
   * 开始会话
   */
  start(session: LearningSession): LearningSession {
    if (session.status !== SessionStatus.NOT_STARTED) {
      throw new DomainError('会话已启动或已完成');
    }
    return {
      ...session,
      status: SessionStatus.IN_PROGRESS,
      startedAt: Date.now(),
    };
  }

  /**
   * 暂停会话
   */
  pause(session: LearningSession, snapshot?: SessionSnapshot): LearningSession {
    if (session.status !== SessionStatus.IN_PROGRESS) {
      throw new DomainError('会话不在进行中，无法暂停');
    }
    return {
      ...session,
      status: SessionStatus.PAUSED,
      pausedAt: Date.now(),
      snapshot: snapshot ?? {
        stepIndex: session.currentStepIndex,
        stepData: {},
        timestamp: Date.now(),
      },
    };
  }

  /**
   * 恢复会话
   */
  resume(session: LearningSession): LearningSession {
    if (session.status !== SessionStatus.PAUSED) {
      throw new DomainError('会话不在暂停状态，无法恢复');
    }
    return {
      ...session,
      status: SessionStatus.IN_PROGRESS,
      pausedAt: undefined,
      snapshot: undefined,
    };
  }

  /**
   * 完成会话
   */
  complete(session: LearningSession): LearningSession {
    if (session.status !== SessionStatus.IN_PROGRESS) {
      throw new DomainError('会话不在进行中，无法完成');
    }
    return {
      ...session,
      status: SessionStatus.COMPLETED,
      completedAt: Date.now(),
    };
  }

  /**
   * 放弃会话
   */
  abandon(session: LearningSession): LearningSession {
    if (
      session.status === SessionStatus.COMPLETED ||
      session.status === SessionStatus.ABANDONED
    ) {
      throw new DomainError('会话已完成或已放弃');
    }
    return {
      ...session,
      status: SessionStatus.ABANDONED,
      completedAt: Date.now(),
    };

  /**
   * 添加证据
   */
  addEvidence(session: LearningSession, evidence: LearningEvidence): LearningSession {
    return {
      ...session,
      evidence: [...session.evidence, evidence],
    };
  }

  /**
   * 前进到下一步
   */
  advanceStep(session: LearningSession): LearningSession {
    return {
      ...session,
      currentStepIndex: session.currentStepIndex + 1,

// src/domain/services/AbilityProfileService.ts

import { AbilityProfile } from '../types/ability-profile';
import { Ability, AbilityLevel, AbilityState } from '../types/ability';
import { LearningEvidence, EvidenceType } from '../types/learning-session';

/**
 * 能力画像领域服务
 */
export class AbilityProfileService {
  /**
   * 创建新的能力画像
   */
  createProfile(userId: string): AbilityProfile {
    return {
      userId,
      states: Object.values(Ability).map((ability) => ({
        ability,
        level: AbilityLevel.NOT_STARTED,
        evidenceCount: 0,
        lastPracticedAt: 0,
      })),
      totalEvidence: [],
      updatedAt: Date.now(),
    };
  }

  /**
   * 根据证据更新能力等级
   */
  updateAbilityLevel(
    profile: AbilityProfile,
    ability: Ability,
    evidence: LearningEvidence[]
  ): AbilityProfile {
    const stateIndex = profile.states.findIndex((s) => s.ability === ability);
    if (stateIndex === -1) {
      return profile;
    }

    const newState = this.calculateLevel(ability, evidence);

    const newStates = [...profile.states];
    newStates[stateIndex] = {
      ...newStates[stateIndex],
      level: newState.level,
      evidenceCount: evidence.length,
      lastPracticedAt: Date.now(),
    };

    return {
      ...profile,
      states: newStates,
      totalEvidence: [...profile.totalEvidence, ...evidence],
      updatedAt: Date.now(),
    };
  }

  /**
   * 计算能力等级
   */
  private calculateLevel(
    ability: Ability,
    evidence: LearningEvidence[]
  ): AbilityState {
    // 能力等级计算规则：
    // NOT_STARTED (0): 无证据
    // AWARE (1): 有操作证据
    // DEVELOPING (2): 有答案证据且正确率 >= 50%
    // MASTERED (3): 有迁移证据且正确率 >= 80%

    const operationCount = evidence.filter(
      (e) => e.type === EvidenceType.OPERATION
    ).length;
    const answerEvidence = evidence.filter(
      (e) => e.type === EvidenceType.ANSWER
    );
    const correctCount = answerEvidence.filter(
      (e) => (e.data as any).isCorrect
    ).length;
    const transferEvidence = evidence.filter(
      (e) => e.type === EvidenceType.TRANSFER
    );
    const successfulTransfer = transferEvidence.filter(
      (e) => (e.data as any).success
    ).length;

    let level = AbilityLevel.NOT_STARTED;

    if (operationCount > 0) {
      level = AbilityLevel.AWARE;
    }

    if (
      answerEvidence.length > 0 &&
      correctCount / answerEvidence.length >= 0.5
    ) {
      level = AbilityLevel.DEVELOPING;
    }

    if (
      transferEvidence.length > 0 &&
      successfulTransfer / transferEvidence.length >= 0.8
    ) {
      level = AbilityLevel.MASTERED;
    }

    return {
      ability,
      level,
      evidenceCount: evidence.length,
      lastPracticedAt: Date.now(),
    };
  }
}

import { ContentRepository } from '../ports/ContentRepository';
import { LearningSession } from '../../domain/types/learning-session';
import { LearningSessionService } from '../../domain/services/LearningSessionService';
import { generateId } from '../../shared/utils';

/**
 * 开始学习会话用例
 */
export class StartLearningSession {
  constructor(
    private sessionRepo: SessionRepository,
    private contentRepo: ContentRepository,
    private sessionService: LearningSessionService
  ) {}

  async execute(userId: string, knowledgePointId: string): Promise<LearningSession> {
    // 1. 检查是否有正在进行的会话
    const activeSession = await this.sessionRepo.findActiveByUserId(userId);
    if (activeSession) {
      // 如果有，且是同一知识点，直接返回
      if (activeSession.knowledgePointId === knowledgePointId) {
        return activeSession;
      }
      // 如果是不同知识点，放弃旧会话
      await this.sessionService.abandon(activeSession);
      await this.sessionRepo.save(activeSession);
    }

    // 2. 获取知识点信息
    const kp = await this.contentRepo.findKnowledgePoint(knowledgePointId);
    if (!kp) {
      throw new Error(`知识点 ${knowledgePointId} 不存在`);
    }

    // 3. 创建新会话
    const session = this.sessionService.createSession(
      generateId(),
      knowledgePointId,
      kp.ability as any
    );

    // 4. 保存并启动
    const startedSession = this.sessionService.start(session);
    await this.sessionRepo.save(startedSession);

    return startedSession;
  }
}

import { SessionRepository } from '../ports/SessionRepository';
import { ContentRepository } from '../ports/ContentRepository';
import { LearningSessionService } from '../../domain/services/LearningSessionService';
import { generateId } from '../../shared/utils';
import { StepType } from '../../domain/types/learning-step';

/**
 * 完成步骤用例
 */
export class CompleteStep {
  constructor(
    private sessionRepo: SessionRepository,
    private contentRepo: ContentRepository,
    private sessionService: LearningSessionService
  ) {}

  async execute(sessionId: string, stepData: Record<string, unknown>): Promise<{
    isLastStep: boolean;
    session: any;
  }> {
    // 1. 获取会话
    const session = await this.sessionRepo.findById(sessionId);
    if (!session) {
      throw new Error(`会话 ${sessionId} 不存在`);
    }

    // 2. 获取步骤信息
    const steps = await this.contentRepo.findSteps(session.knowledgePointId);
    const currentStep = steps[session.currentStepIndex];

    if (!currentStep) {
      throw new Error(`步骤 ${session.currentStepIndex} 不存在`);
    }

    // 3. 记录操作证据
    await this.sessionRepo.saveEvidence(sessionId, {
      id: generateId(),
      stepId: currentStep.id,
      type: 'operation' as any,
      data: {
        type: 'operation',
        interactionKind: (currentStep.content as any).interaction?.kind ?? 'tap',
        result: stepData,
        attempts: 1,
      },
      timestamp: Date.now(),
    });

    // 4. 判断是否是最后一步
    const isLastStep = session.currentStepIndex >= steps.length - 1;

    // 5. 更新会话
    let updatedSession: any;
    if (isLastStep) {
      updatedSession = this.sessionService.complete(session);
    } else {
      updatedSession = this.sessionService.advanceStep(session);
    }

    await this.sessionRepo.save(updatedSession);

    return { isLastStep, session: updatedSession };
  }
}

import type { ConcreteContent } from '../../domain/types/learning-step';

interface ConcreteStepProps {
  /** 步骤内容 */
  content: ConcreteContent;
  /** 步骤名称 */
  stepName: string;
  /** 步骤索引 */
  stepIndex: number;
  /** 提示层级 */
  hintLevels: string[];
  /** 完成回调 */
  onComplete: (data: Record<string, unknown>) => void;
  /** 请求提示回调 */
  onHint?: () => void;

/**
 * 具象操作步骤组件
 * - 接收 props 中的数据和回调
 * - 不直接调用 store 或 service
 */
export const ConcreteStep: React.FC<ConcreteStepProps> = ({
  content,
  stepName,
  stepIndex,
  hintLevels,
  onComplete,
  onHint,
}) => {
  const [hintsUsed, setHintsUsed] = useState(0);
  const [feedback, setFeedback] = useState<{ type: 'success' | 'hint'; message: string } | null>(null);
  const [isComplete, setIsComplete] = useState(false);

  // 处理可视化交互完成
  const handleVisualComplete = useCallback(() => {
    setIsComplete(true);
    setFeedback({
      type: 'success',
      message: content.interaction.target + ' 完成！',
    });
  }, [content]);

  // 处理完成按钮点击
  const handleComplete = useCallback(() => {
    if (!isComplete) return;
    onComplete({ completed: true, hintsUsed });
  }, [isComplete, hintsUsed, onComplete]);

  // 处理提示请求
  const handleHint = useCallback(() => {
    if (hintsUsed < hintLevels.length) {
      setHintsUsed((prev) => prev + 1);
      setFeedback({
        type: 'hint',
        message: hintLevels[hintsUsed],
      });
      onHint?.();
    }
  }, [hintsUsed, hintLevels, onHint]);

  // 从内容中提取可视化配置
  const visualConfig = content.interaction.visualConfig ?? {};

  return (
    <div className="step concrete-step">
      <StepHeader
        stepName={stepName}
        stepIndex={stepIndex}
        emoji={content.scenario.emoji}
      />

      <div className="step-content">
        <div className="scenario-card">
          <h3>{content.scenario.title}</h3>
          <p>{content.scenario.description}</p>
        </div>

        <div className="task-card">
          <span className="task-label">任务</span>
          <p>{content.interaction.target}</p>
        </div>

        <div className="visual-area">
          <SplitCircle
            denominator={(visualConfig as any).denominator ?? 2}
            highlighted={(visualConfig as any).highlighted ?? []}
            onComplete={handleVisualComplete}
          />
        </div>

        {feedback && (
          <StepFeedback type={feedback.type} message={feedback.message} />
        )}
      </div>

      <div className="step-actions">
        {hintsUsed < hintLevels.length && (
          <Button variant="ghost" onClick={handleHint}>
            提示 ({hintLevels.length - hintsUsed})
          </Button>
        )}
        <Button
          variant="primary"
          onClick={handleComplete}
          disabled={!isComplete}
        >
          {isComplete ? '完成' : '先完成任务'}
        </Button>
      </div>
    </div>
  );
};

const ConjectureContentSchema = z.object({

// src/stores/sessionStore.ts

import { create } from 'zustand';
import { subscribeWithSelector } from 'zustand/middleware';
import type { LearningSession } from '../domain/types/learning-session';
import type { LearningStep } from '../domain/types/learning-step';

interface SessionState {
  /** 当前会话 */
  session: LearningSession | null;
  /** 当前步骤 */
  currentStep: LearningStep | null;
  /** 步骤列表 */
  steps: LearningStep[];
  /** 加载状态 */
  isLoading: boolean;
  /** 错误信息 */
  error: string | null;
  /** 是否显示完成页 */
  showComplete: boolean;
}

interface SessionActions {
  /** 开始学习会话 */
  startSession: (session: LearningSession, steps: LearningStep[]) => void;
  /** 完成当前步骤 */
  completeStep: (data: Record<string, unknown>) => void;
  /** 请求提示 */
  requestHint: () => void;
  /** 暂停会话 */
  pauseSession: () => void;
  /** 恢复会话 */
  resumeSession: () => void;
  /** 放弃会话 */
  abandonSession: () => void;
  /** 重置状态 */
  reset: () => void;
  /** 清除错误 */
  clearError: () => void;
}

type SessionStore = SessionState & SessionActions;

const initialState: SessionState = {
  session: null,
  currentStep: null,
  steps: [],
  isLoading: false,
  error: null,
  showComplete: false,
};

export const useSessionStore = create<SessionStore>()(
  subscribeWithSelector((set, get) => ({
    ...initialState,

    startSession: (session, steps) => {
      set({
        session,
        steps,
        currentStep: steps[0] ?? null,
        isLoading: false,
        error: null,
        showComplete: false,
      });
    },

    completeStep: (data) => {
      const { session, steps } = get();
      if (!session) return;

      const nextIndex = session.currentStepIndex + 1;
      const isLastStep = nextIndex >= steps.length;

      if (isLastStep) {
        set({
          session: {
            ...session,
            currentStepIndex: nextIndex,
            status: 'completed' as any,
            completedAt: Date.now(),
          },
          showComplete: true,
        });
      } else {
        set({
          session: {
            ...session,
            currentStepIndex: nextIndex,
          },
          currentStep: steps[nextIndex],
        });
      }
    },

    requestHint: () => {
      // 提示逻辑由步骤组件处理
    },

    pauseSession: () => {
      const { session } = get();
      if (!session) return;

      set({
        session: {
          ...session,
          status: 'paused' as any,
          pausedAt: Date.now(),
        },
      });
    },

    resumeSession: () => {
      const { session } = get();
      if (!session) return;

      set({
        session: {
          ...session,
          status: 'in-progress' as any,
          pausedAt: undefined,
        },
      });
    },

    abandonSession: () => {
      const { session } = get();
      if (!session) return;

      set({
        session: {
          ...session,
          status: 'abandoned' as any,
          completedAt: Date.now(),
        },
      });
    },

    reset: () => {
      set(initialState);
    },

    clearError: () => {
      set({ error: null });
    },
  }))
);
```
