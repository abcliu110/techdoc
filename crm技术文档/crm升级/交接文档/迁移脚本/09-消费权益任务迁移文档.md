## 1. 新建表：`crm_consume_benefit_task`（消费权益任务）

**文件名**：`V20260513_crm_consume_benefit_task.sql`
**目标库**：CRM 数据库

### 1.1 建表语句

```sql
CREATE TABLE IF NOT EXISTS `crm_consume_benefit_task` (
  `pid` BIGINT NOT NULL AUTO_INCREMENT COMMENT '物理主键',
  `company_id` BIGINT NOT NULL COMMENT '商户ID',
  `shop_id` BIGINT DEFAULT NULL COMMENT '门店ID',
  `lmnid` BIGINT NOT NULL COMMENT '业务逻辑主键',
  `origin_source` TINYINT NOT NULL COMMENT '消费权益来源：1-POS，2-订单',
  `origin_lifecycle_id` BIGINT NOT NULL COMMENT '来源生命周期ID，订单侧使用订单LID',
  `event_type` TINYINT NOT NULL COMMENT '事件类型：1-发放，2-回收',
  `out_order_id` VARCHAR(64) NOT NULL COMMENT '外部订单号',
  `card_lid` BIGINT NOT NULL COMMENT '会员卡LID',
  `card_no` VARCHAR(64) DEFAULT NULL COMMENT '会员卡号',
  `business_key` VARCHAR(255) NOT NULL COMMENT '业务幂等键',
  `event_payload` LONGTEXT DEFAULT NULL COMMENT '订单侧消费权益事件JSON快照，用于后台补偿重放',
  `status_` TINYINT NOT NULL COMMENT '状态：0-待处理，1-回收中，2-已回收，3-可重试，4-最终失败',
  `executed_time` DATETIME DEFAULT NULL COMMENT '执行时间',
  `last_error` VARCHAR(1024) DEFAULT NULL COMMENT '最近错误信息',
  `revision` INT NOT NULL DEFAULT 0 COMMENT '数据版本',
  `created_by` VARCHAR(64) DEFAULT NULL COMMENT '创建人',
  `created_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_by` VARCHAR(64) DEFAULT NULL COMMENT '更新人',
  `updated_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '逻辑删除(0-否,1-是)',
  PRIMARY KEY (`pid`),
  UNIQUE KEY `uk_crm_consume_benefit_task_business_key_deleted` (`business_key`, `deleted`),
  UNIQUE KEY `uk_crm_consume_benefit_task_lmnid` (`lmnid`),
  KEY `idx_crm_consume_benefit_task_mid` (`company_id`),
  KEY `idx_crm_consume_benefit_task_status` (`status_`),
  KEY `idx_crm_consume_benefit_task_origin` (`origin_source`, `origin_lifecycle_id`, `event_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='CRM消费权益任务';
```

### 1.2 字段说明

| 字段 | 类型 | 说明 |
|------|------|------|
| `origin_source` | TINYINT | 消费权益来源：1-POS，2-订单 |
| `origin_lifecycle_id` | BIGINT | 来源生命周期 ID，订单侧使用订单 LID |
| `event_type` | TINYINT | 事件类型：1-发放，2-回收 |
| `business_key` | VARCHAR(255) | 业务幂等键 |
| `event_payload` | LONGTEXT | 事件 JSON 快照，用于后台补偿重放 |
| `status_` | TINYINT | 状态：0-待处理，1-回收中，2-已回收，3-可重试，4-最终失败 |

## 2. 新建表：`order_consume_benefit_outbox`（订单消费权益 Outbox）

**文件名**：`V20260513_order_consume_benefit_outbox.sql`
**目标库**：订单数据库

### 2.1 建表语句

```sql
CREATE TABLE IF NOT EXISTS `order_consume_benefit_outbox` (
  `pid` BIGINT NOT NULL AUTO_INCREMENT COMMENT '物理主键',
  `mid` BIGINT NOT NULL COMMENT '商户ID',
  `sid` BIGINT NOT NULL COMMENT '门店ID',
  `lid` BIGINT NOT NULL COMMENT '业务逻辑主键',
  `origin_source` TINYINT NOT NULL COMMENT '消费权益来源：1-POS，2-订单',
  `origin_lifecycle_id` BIGINT NOT NULL COMMENT '来源生命周期ID，订单侧使用订单LID',
  `event_type` TINYINT NOT NULL COMMENT '事件类型：1-发放，2-回收',
  `out_order_id` VARCHAR(64) NOT NULL COMMENT '外部订单号',
  `card_lid` BIGINT NOT NULL COMMENT '会员卡LID',
  `business_key` VARCHAR(255) NOT NULL COMMENT '业务幂等键',
  `payload` LONGTEXT NOT NULL COMMENT '待投递事件载荷',
  `status_` TINYINT NOT NULL COMMENT '状态：0-待投递，1-已投递，2-可重试',
  `dispatched_time` DATETIME DEFAULT NULL COMMENT '投递时间',
  `last_error` VARCHAR(1024) DEFAULT NULL COMMENT '最近错误信息',
  `revision` INT NOT NULL DEFAULT 0 COMMENT '数据版本',
  `created_by` VARCHAR(64) DEFAULT NULL COMMENT '创建人',
  `created_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_by` VARCHAR(64) DEFAULT NULL COMMENT '更新人',
  `updated_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '逻辑删除(0-否,1-是)',
  PRIMARY KEY (`pid`),
  UNIQUE KEY `uk_order_consume_benefit_outbox_business_key_deleted` (`business_key`, `deleted`),
  UNIQUE KEY `uk_order_consume_benefit_outbox_lid` (`lid`),
  KEY `idx_order_consume_benefit_outbox_mid_sid` (`mid`, `sid`),
  KEY `idx_order_consume_benefit_outbox_status` (`status_`),
  KEY `idx_order_consume_benefit_outbox_origin` (`origin_source`, `origin_lifecycle_id`, `event_type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单消费权益outbox';
```
