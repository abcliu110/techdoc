## 1. 新建表：`crm_points_rule`（积分权益规则主表）

**文件名**：`2026-04-17_积分权益规则表.sql`
**目标库**：CRM 数据库

### 1.1 建表语句

```sql
CREATE TABLE `crm_points_rule` (
  `pid` BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '物理编号',
  `mid` BIGINT NOT NULL COMMENT '商户ID',
  `sid` BIGINT NULL COMMENT '门店ID；兼容字段，积分权益规则当前不按门店隔离，保存时为空',
  `lid` BIGINT NOT NULL COMMENT '逻辑编号(雪花算法)',
  `name` VARCHAR(100) COMMENT '规则名称；当前接口层未开放编辑，通常为空，预留给后台运营展示',
  `plan_lid` BIGINT NOT NULL COMMENT '所属会员方案逻辑编号',
  `points_rule_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '积分规则全局开关:0-否,1-是',
  -- 积分获取规则
  `earning_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '是否启用消费送积分:0-否,1-是',
  `gift_amount_earn_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '储值赠送参与积分:0-关闭,1-开启',
  `earning_mode` TINYINT COMMENT '积分模式:1-按支付方式实收积分,2-按商品实收积分',
  `earning_product_scope_type` TINYINT COMMENT '获取积分适用商品范围:1-全部商品,2-指定商品参与,3-指定商品不参与',
  `earning_specified_product_lids` JSON COMMENT '获取积分指定商品ID列表(JSON数组)',
  `level_rates` JSON COMMENT '会员等级赠分规则(JSON数组)',
  `single_earn_limit_type` TINYINT COMMENT '单笔获取上限类型:1-不限制,2-限制',
  `single_earn_limit_value` DECIMAL(18,2) COMMENT '单笔获取上限积分值',
  `bonus_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '多倍积分总开关:0-否,1-是',
  `birthday_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '生日多倍积分开关:0-否,1-是',
  `birthday_period` TINYINT COMMENT '生日范围:1-当天,2-当周,3-当月',
  `birthday_multiplier` DECIMAL(5,2) COMMENT '生日积分倍数',
  `member_day_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '会员日多倍积分开关:0-否,1-是',
  `member_day_period` TINYINT COMMENT '会员日周期:1-每周,2-每月',
  `member_day_days_of_week` JSON COMMENT '会员日按周生效星期',
  `member_day_days_of_month` JSON COMMENT '会员日按月生效日期',
  `member_day_multiplier` DECIMAL(5,2) COMMENT '会员日积分倍数',
  `available_cycle` TINYINT COMMENT '获取积分可用周期:1-按日,2-按周,3-按月',
  `available_days_of_week` JSON COMMENT '获取积分按周生效星期',
  `available_days_of_month` JSON COMMENT '获取积分按月生效日期',
  `available_time_type` TINYINT COMMENT '获取积分可用时段类型:1-全天,2-指定时段',
  `available_time_slots` JSON COMMENT '获取积分指定时段(JSON数组)',
  `wecom_exclusive_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '企微会员专享开关:0-否,1-是',
  `wecom_exclusive_type` TINYINT COMMENT '企微会员专享类型:1-好友或群客,2-好友,3-群客,4-好友且群客',
  `order_scene_limit` JSON COMMENT '获取积分订单场景限制',
  `order_channel_limit` JSON COMMENT '获取积分订单渠道限制',
  -- 积分抵扣规则
  `deduction_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '是否启用积分抵现:0-否,1-是',
  `deduction_rate` DECIMAL(18,4) COMMENT '积分抵扣比例(X积分=1元)',
  `min_deduction_type` TINYINT COMMENT '起扣积分限制类型:1-不限制,2-限制',
  `min_deduction_value` DECIMAL(18,2) COMMENT '起扣积分数',
  `deduction_multiple_type` TINYINT COMMENT '抵扣倍数类型:1-不限制,2-限制',
  `deduction_multiple_value` INT COMMENT '抵扣整数倍数值',
  `deduction_rounding` TINYINT COMMENT '抵扣金额取整:1-不取整,2-向下取整',
  `deduction_ceiling_type` TINYINT COMMENT '抵扣上限类型:1-不限制,2-固定积分,3-账单比例',
  `deduction_ceiling_points` DECIMAL(18,2) COMMENT '固定积分上限',
  `deduction_ceiling_ratio` DECIMAL(5,2) COMMENT '账单比例上限，单位%',
  `deduct_cycle` TINYINT COMMENT '积分抵现可用周期:1-按日,2-按周,3-按月',
  `deduct_days_of_week` JSON COMMENT '积分抵现按周生效星期',
  `deduct_days_of_month` JSON COMMENT '积分抵现按月生效日期',
  `deduct_time_type` TINYINT COMMENT '积分抵现可用时段类型:1-全天,2-指定时段',
  `deduct_time_slots` JSON COMMENT '积分抵现指定时段(JSON数组)',
  `product_limit_enabled` TINYINT NOT NULL DEFAULT 0 COMMENT '抵现商品限制开关:0-否,1-是',
  `product_scope_type` TINYINT COMMENT '抵现商品范围',
  `deduct_specified_product_lids` JSON COMMENT '抵现指定商品ID列表(JSON数组)',
  `deduct_pos_category_lids` JSON COMMENT '抵现POS分类ID列表(JSON数组)',
  `deduct_miniapp_category_lids` JSON COMMENT '抵现小程序分类ID列表(JSON数组)',
  `deduct_exclude_product_lids` JSON COMMENT '抵现排除商品ID列表(JSON数组)',
  `deduct_scene_limit` JSON COMMENT '抵现订单场景限制',
  `deduct_channel_limit` JSON COMMENT '抵现订单渠道限制',
  -- 积分有效期规则
  `validity_mode` TINYINT NOT NULL DEFAULT 1 COMMENT '积分有效期模式:1-永久有效,2-每年固定日期清零',
  `validity_fixed_month` TINYINT COMMENT '每年固定日期清零月份，validity_mode=2 时有值，范围1-12',
  `validity_fixed_day` TINYINT COMMENT '每年固定日期清零日期，validity_mode=2 时有值，2月最多28日',
  `validity_next_clear_time` DATETIME COMMENT '下一次固定日期积分清零时间，仅固定日期模式有值',
  `validity_notify_enabled` TINYINT COMMENT '清零前消息提醒开关:0-不提醒,1-提醒',
  `validity_remind_days` INT COMMENT '距清零提前提醒天数，validity_notify_enabled=1 时有值',
  -- 其他
  `rule_description` VARCHAR(600) COMMENT '积分规则公示说明',
  `created_by` VARCHAR(100) COMMENT '创建人',
  `created_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_by` VARCHAR(100) COMMENT '更新人',
  `updated_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `deleted` TINYINT NOT NULL DEFAULT 0 COMMENT '逻辑删除(0-否,1-是)',
  `revision` INT NOT NULL DEFAULT 0 COMMENT '乐观锁版本号'
) COMMENT='积分权益规则主表';

CREATE INDEX idx_mid ON crm_points_rule(mid);
CREATE UNIQUE INDEX uk_lid ON crm_points_rule(lid);
CREATE INDEX idx_plan_lid ON crm_points_rule(plan_lid);
CREATE UNIQUE INDEX uk_mid_plan_lid ON crm_points_rule(mid, plan_lid);
CREATE INDEX idx_crm_points_rule_validity_next_clear_time ON crm_points_rule(validity_mode, points_rule_enabled, validity_next_clear_time);
```

### 1.2 积分获取规则字段

| 字段 | 类型 | 说明 |
|------|------|------|
| `earning_enabled` | TINYINT | 消费送积分开关：0-否，1-是 |
| `gift_amount_earn_enabled` | TINYINT | 储值赠送参与积分：0-关闭，1-开启 |
| `earning_mode` | TINYINT | 积分模式：1-按支付方式实收，2-按商品实收 |
| `level_rates` | JSON | 会员等级赠分规则 |
| `single_earn_limit_type/value` | TINYINT/DECIMAL | 单笔获取上限 |
| `bonus_enabled` | TINYINT | 多倍积分总开关 |
| `birthday_*` | TINYINT/DECIMAL | 生日多倍积分 |
| `member_day_*` | TINYINT/DECIMAL | 会员日多倍积分 |
| `order_scene_limit` | JSON | 订单场景限制 |
| `order_channel_limit` | JSON | 订单渠道限制 |

### 1.3 积分抵扣规则字段

| 字段 | 类型 | 说明 |
|------|------|------|
| `deduction_enabled` | TINYINT | 积分抵现开关：0-否，1-是 |
| `deduction_rate` | DECIMAL | 抵扣比例（如每100积分抵1元） |
| `deduct_ceiling_type` | TINYINT | 抵扣上限：1-不限制，2-固定积分，3-账单比例 |
| `product_limit_enabled` | TINYINT | 抵现商品限制开关 |

### 1.4 积分有效期规则字段

| 字段 | 类型 | 说明 |
|------|------|------|
| `validity_mode` | TINYINT | 1-永久有效，2-每年固定日期清零 |
| `validity_fixed_month` | TINYINT | 清零月份（1-12） |
| `validity_fixed_day` | TINYINT | 清零日期（2月最多28日） |
| `validity_next_clear_time` | DATETIME | 下一次清零时间 |
| `validity_notify_enabled` | TINYINT | 清零前消息提醒开关 |

## 2. 新增字段：`crm_points_rule` 积分有效期字段

**文件名**：`V20260604__crm_points_rule_validity_columns.sql`
**目标库**：CRM 数据库

```sql
ALTER TABLE `crm_points_rule`
  ADD COLUMN `validity_mode` TINYINT NOT NULL DEFAULT 1 COMMENT '积分有效期模式:1-永久有效,2-每年固定日期清零' AFTER `deduct_channel_limit`,
  ADD COLUMN `validity_fixed_month` TINYINT NULL COMMENT '每年固定日期清零月份，validity_mode=2 时有值，范围1-12' AFTER `validity_mode`,
  ADD COLUMN `validity_fixed_day` TINYINT NULL COMMENT '每年固定日期清零日期，validity_mode=2 时有值，2月最多28日' AFTER `validity_fixed_month`,
  ADD COLUMN `validity_next_clear_time` DATETIME NULL COMMENT '下一次固定日期积分清零时间，仅固定日期模式有值' AFTER `validity_fixed_day`,
  ADD COLUMN `validity_notify_enabled` TINYINT NULL COMMENT '清零前消息提醒开关:0-不提醒,1-提醒' AFTER `validity_next_clear_time`,
  ADD COLUMN `validity_remind_days` INT NULL COMMENT '距清零提前提醒天数，validity_notify_enabled=1 时有值' AFTER `validity_notify_enabled`;

ALTER TABLE `crm_points_rule`
  ADD INDEX `idx_crm_points_rule_validity_next_clear_time`
    (`validity_mode`, `points_rule_enabled`, `validity_next_clear_time`);
```
