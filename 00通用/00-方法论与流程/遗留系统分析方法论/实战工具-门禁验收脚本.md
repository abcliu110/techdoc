# 深度展开4：门禁验收的可执行脚本

## 为什么需要自动化门禁

### 人工验收的问题

```text
问题1：主观判断
"这个模块分析得差不多了" → 实际上漏了30%的规则

问题2：放水倾向
"时间紧，先过吧" → 后续问题滚雪球

问题3：标准不一致
不同评审人标准不同 → 质量波动大

问题4：无法追溯
事后发现问题，无法定位是哪个环节放水
```

### 自动化门禁的价值

```text
价值1：客观标准
脚本退出码 = 0/2/3/4/5，无争议

价值2：可重复
任何时候重跑，结果一致

价值3：可追溯
检查点记录脚本退出码，责任清晰

价值4：强制执行
非0退出码直接阻断后续阶段
```

## G1门禁脚本实现

### 脚本1：check-level1-skeleton.ps1（Windows）

```powershell
<#
.SYNOPSIS
    检查Level 1骨架完整性

.DESCRIPTION
    验证所有业务域的Level 1骨架文件是否齐全且有实质内容
    退出码：
    0 = 检查通过
    2 = 骨架文件缺失
    3 = Level 2孤立（所属业务域无Level 1）
    4 = 清单不一致
    5 = 内容为占位符（G1未完成）

.PARAMETER RootPath
    系统分析产出根目录

.EXAMPLE
    .\check-level1-skeleton.ps1 -RootPath "D:\mywork\techdoc\00通用\00-方法论与流程\分析系统的方法论\dist1\系统分析产出"
#>

param(
    [Parameter(Mandatory=$true)]
    [string]$RootPath
)

# 设置输出编码为UTF-8
$OutputEncoding = [System.Text.Encoding]::UTF8
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8

Write-Host "========================================" -ForegroundColor Cyan
Write-Host "Level 1 骨架完整性检查" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host ""

# 检查根目录是否存在
if (-not (Test-Path $RootPath)) {
    Write-Host "[错误] 根目录不存在: $RootPath" -ForegroundColor Red
    exit 1
}

# 读取Level 0业务域清单
$level0Path = Join-Path $RootPath "Level 0"
$moduleFile = Join-Path $level0Path "02-模块分级.md"

if (-not (Test-Path $moduleFile)) {
    Write-Host "[错误] 业务域清单文件不存在: $moduleFile" -ForegroundColor Red
    exit 1
}

# 解析业务域清单
$content = Get-Content $moduleFile -Encoding UTF8 -Raw
$domains = @()

# 匹配业务域表格行（简化版，实际需要更严格的正则）
$lines = $content -split "`n"
$inTable = $false
foreach ($line in $lines) {
    if ($line -match '^\s*\|\s*业务域\d+') {
        $inTable = $true
    }
    if ($inTable -and $line -match '^\s*\|\s*业务域(\d+)[^\|]*\|\s*([^\|]+)\|\s*\(([^\)]+)\)') {
        $domains += @{
            Number = $matches[1]
            Name = $matches[2].Trim()
            Label = $matches[3].Trim()
        }
    }
}

Write-Host "[信息] 从Level 0读取到 $($domains.Count) 个业务域" -ForegroundColor Green
Write-Host ""

# 检查Level 1目录
$level1Path = Join-Path $RootPath "Level 1-业务域层"
if (-not (Test-Path $level1Path)) {
    Write-Host "[错误] Level 1目录不存在: $level1Path" -ForegroundColor Red
    exit 2
}

# 检查每个业务域的骨架
$missingDomains = @()
$missingSkeleton = @()
$placeholderDomains = @()

foreach ($domain in $domains) {
    $domainDir = Join-Path $level1Path "业务域$($domain.Number)-$($domain.Label)($($domain.Name))"
    
    Write-Host "[检查] 业务域$($domain.Number): $($domain.Name)" -ForegroundColor Yellow
    
    # 检查目录是否存在
    if (-not (Test-Path $domainDir)) {
        Write-Host "  [失败] 目录不存在: $domainDir" -ForegroundColor Red
        $missingDomains += $domain.Number
        continue
    }
    
    # 检查三个必需文件
    $requiredFiles = @(
        "01-业务域概览.md",
        "02-上下文清单.md",
        "03-业务域总结.md"
    )
    
    $hasAllFiles = $true
    $hasSkeleton = $true
    $hasContent = $true
    
    foreach ($file in $requiredFiles) {
        $filePath = Join-Path $domainDir $file
        
        if (-not (Test-Path $filePath)) {
            Write-Host "  [失败] 文件缺失: $file" -ForegroundColor Red
            $missingSkeleton += "$($domain.Number)-$file"
            $hasAllFiles = $false
            continue
        }
        
        # 检查文件内容
        $fileContent = Get-Content $filePath -Encoding UTF8 -Raw
        
        # 检查skeleton标记
        if ($fileContent -notmatch '<!--\s*skeleton:\s*do\s*not\s*delete\s*-->') {
            Write-Host "  [警告] 文件缺少skeleton标记: $file" -ForegroundColor Yellow
            $hasSkeleton = $false
        }
        
        # 检查是否有实质内容（排除占位符）
        # 移除skeleton标记后的内容
        $contentAfterMarker = $fileContent -replace '(?s).*<!--\s*skeleton:\s*do\s*not\s*delete\s*-->', ''
        
        # 检查是否全是占位符
        $placeholderPatterns = @(
            '^\s*-\s*$',           # 只有 "-"
            '^\s*待补充\s*$',      # 只有 "待补充"
            '^\s*待分析\s*$',      # 只有 "待分析"
            '^\s*TBD\s*$',         # 只有 "TBD"
            '^\s*TODO\s*$',        # 只有 "TODO"
            '^\s*\|\s*-+\s*\|\s*$' # 空表格行
        )
        
        $hasRealContent = $false
        $contentLines = $contentAfterMarker -split "`n"
        foreach ($contentLine in $contentLines) {
            $line = $contentLine.Trim()
            if ($line.Length -eq 0) { continue }
            if ($line -match '^#') { continue }  # 跳过标题
            if ($line -match '^\|.*\|$' -and $line -match '^[\|\s-]+$') { continue }  # 跳过表格分隔线
            
            $isPlaceholder = $false
            foreach ($pattern in $placeholderPatterns) {
                if ($line -match $pattern) {
                    $isPlaceholder = $true
                    break
                }
            }
            
            if (-not $isPlaceholder) {
                $hasRealContent = $true
                break
            }
        }
        
        if (-not $hasRealContent) {
            Write-Host "  [失败] 文件内容全为占位符: $file" -ForegroundColor Red
            $hasContent = $false
        }
    }
    
    if ($hasAllFiles -and $hasSkeleton -and $hasContent) {
        Write-Host "  [通过] 骨架完整且有实质内容" -ForegroundColor Green
    } elseif ($hasAllFiles -and $hasSkeleton -and -not $hasContent) {
        Write-Host "  [失败] 骨架齐全但内容为占位符" -ForegroundColor Red
        $placeholderDomains += $domain.Number
    }
    
    Write-Host ""
}

# 检查Level 2孤立任务
$level2Path = Join-Path $RootPath "Level 2"
$orphanContexts = @()

if (Test-Path $level2Path) {
    $level2Dirs = Get-ChildItem $level2Path -Directory
    
    foreach ($contextDir in $level2Dirs) {
        $contextName = $contextDir.Name
        
        # 提取业务域标识（假设格式：上下文xxx-yyy(domain)）
        if ($contextName -match '\(([^\)]+)\)') {
            $domainLabel = $matches[1]
            
            # 检查该业务域是否在Level 0清单中
            $domainExists = $domains | Where-Object { $_.Label -eq $domainLabel }
            
            if (-not $domainExists) {
                Write-Host "[警告] Level 2上下文孤立: $contextName (所属业务域 $domainLabel 不在Level 0清单中)" -ForegroundColor Yellow
                $orphanContexts += $contextName
            }
            
            # 检查Level 1骨架是否存在
            $level1Exists = $domains | Where-Object { $_.Label -eq $domainLabel } | ForEach-Object {
                $domainDir = Join-Path $level1Path "业务域$($_.Number)-$($_.Label)($($_.Name))"
                Test-Path $domainDir
            }
            
            if ($level1Exists -contains $false) {
                Write-Host "[警告] Level 2上下文孤立: $contextName (所属业务域的Level 1骨架不存在)" -ForegroundColor Yellow
                $orphanContexts += $contextName
            }
        }
    }
}

# 汇总结果
Write-Host "========================================" -ForegroundColor Cyan
Write-Host "检查结果汇总" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host ""

Write-Host "业务域总数: $($domains.Count)" -ForegroundColor White
Write-Host "目录缺失: $($missingDomains.Count)" -ForegroundColor $(if ($missingDomains.Count -eq 0) { "Green" } else { "Red" })
Write-Host "骨架缺失: $($missingSkeleton.Count)" -ForegroundColor $(if ($missingSkeleton.Count -eq 0) { "Green" } else { "Red" })
Write-Host "内容占位符: $($placeholderDomains.Count)" -ForegroundColor $(if ($placeholderDomains.Count -eq 0) { "Green" } else { "Red" })
Write-Host "Level 2孤立: $($orphanContexts.Count)" -ForegroundColor $(if ($orphanContexts.Count -eq 0) { "Green" } else { "Red" })
Write-Host ""

# 决定退出码
if ($missingSkeleton.Count -gt 0 -or $missingDomains.Count -gt 0) {
    Write-Host "[结论] 检查不通过：骨架文件缺失（退出码 2）" -ForegroundColor Red
    exit 2
}

if ($orphanContexts.Count -gt 0) {
    Write-Host "[结论] 检查不通过：Level 2孤立（退出码 3）" -ForegroundColor Red
    exit 3
}

# 检查业务域编号连续性
$numbers = $domains | ForEach-Object { [int]$_.Number } | Sort-Object
for ($i = 0; $i -lt $numbers.Count; $i++) {
    if ($numbers[$i] -ne ($i + 1)) {
        Write-Host "[结论] 检查不通过：业务域编号不连续（退出码 4）" -ForegroundColor Red
        exit 4
    }
}

if ($placeholderDomains.Count -gt 0) {
    Write-Host "[结论] 检查不通过：内容为占位符，G1未完成（退出码 5）" -ForegroundColor Red
    Write-Host "未完成的业务域: $($placeholderDomains -join ', ')" -ForegroundColor Yellow
    exit 5
}

Write-Host "[结论] 检查通过：Level 1骨架完整且有实质内容（退出码 0）" -ForegroundColor Green
exit 0
```

### 脚本2：check-level1-skeleton.sh（Linux/macOS）

```bash
#!/bin/bash

# check-level1-skeleton.sh
# 检查Level 1骨架完整性
#
# 退出码：
# 0 = 检查通过
# 2 = 骨架文件缺失
# 3 = Level 2孤立
# 4 = 清单不一致
# 5 = 内容为占位符

set -e

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
CYAN='\033[0;36m'
NC='\033[0m' # No Color

# 参数检查
if [ $# -ne 1 ]; then
    echo -e "${RED}[错误] 用法: $0 <系统分析产出根目录>${NC}"
    exit 1
fi

ROOT_PATH="$1"

echo -e "${CYAN}========================================${NC}"
echo -e "${CYAN}Level 1 骨架完整性检查${NC}"
echo -e "${CYAN}========================================${NC}"
echo ""

# 检查根目录
if [ ! -d "$ROOT_PATH" ]; then
    echo -e "${RED}[错误] 根目录不存在: $ROOT_PATH${NC}"
    exit 1
fi

# 读取业务域清单
MODULE_FILE="$ROOT_PATH/Level 0/02-模块分级.md"

if [ ! -f "$MODULE_FILE" ]; then
    echo -e "${RED}[错误] 业务域清单文件不存在: $MODULE_FILE${NC}"
    exit 1
fi

# 解析业务域（简化版）
DOMAIN_COUNT=$(grep -c "^| 业务域[0-9]" "$MODULE_FILE" || true)
echo -e "${GREEN}[信息] 从Level 0读取到 $DOMAIN_COUNT 个业务域${NC}"
echo ""

# 检查Level 1目录
LEVEL1_PATH="$ROOT_PATH/Level 1-业务域层"

if [ ! -d "$LEVEL1_PATH" ]; then
    echo -e "${RED}[错误] Level 1目录不存在: $LEVEL1_PATH${NC}"
    exit 2
fi

# 初始化计数器
MISSING_COUNT=0
PLACEHOLDER_COUNT=0
ORPHAN_COUNT=0

# 检查每个业务域目录
for domain_dir in "$LEVEL1_PATH"/业务域*; do
    if [ ! -d "$domain_dir" ]; then
        continue
    fi
    
    domain_name=$(basename "$domain_dir")
    echo -e "${YELLOW}[检查] $domain_name${NC}"
    
    # 检查三个必需文件
    has_all=true
    has_content=true
    
    for file in "01-业务域概览.md" "02-上下文清单.md" "03-业务域总结.md"; do
        file_path="$domain_dir/$file"
        
        if [ ! -f "$file_path" ]; then
            echo -e "  ${RED}[失败] 文件缺失: $file${NC}"
            has_all=false
            MISSING_COUNT=$((MISSING_COUNT + 1))
            continue
        fi
        
        # 检查skeleton标记
        if ! grep -q "skeleton: do not delete" "$file_path"; then
            echo -e "  ${YELLOW}[警告] 文件缺少skeleton标记: $file${NC}"
        fi
        
        # 检查是否有实质内容（简化版）
        content_lines=$(sed -n '/skeleton: do not delete/,$p' "$file_path" | \
                       grep -v "^#" | \
                       grep -v "^$" | \
                       grep -v "^-$" | \
                       grep -v "待补充" | \
                       grep -v "待分析" | \
                       grep -v "^|.*-.*|$" || true)
        
        if [ -z "$content_lines" ]; then
            echo -e "  ${RED}[失败] 文件内容全为占位符: $file${NC}"
            has_content=false
        fi
    done
    
    if $has_all && $has_content; then
        echo -e "  ${GREEN}[通过] 骨架完整且有实质内容${NC}"
    elif $has_all && ! $has_content; then
        echo -e "  ${RED}[失败] 骨架齐全但内容为占位符${NC}"
        PLACEHOLDER_COUNT=$((PLACEHOLDER_COUNT + 1))
    fi
    
    echo ""
done

# 汇总结果
echo -e "${CYAN}========================================${NC}"
echo -e "${CYAN}检查结果汇总${NC}"
echo -e "${CYAN}========================================${NC}"
echo ""

echo "业务域总数: $DOMAIN_COUNT"
echo -e "骨架缺失: ${RED}$MISSING_COUNT${NC}"
echo -e "内容占位符: ${RED}$PLACEHOLDER_COUNT${NC}"
echo -e "Level 2孤立: ${RED}$ORPHAN_COUNT${NC}"
echo ""

# 决定退出码
if [ $MISSING_COUNT -gt 0 ]; then
    echo -e "${RED}[结论] 检查不通过：骨架文件缺失（退出码 2）${NC}"
    exit 2
fi

if [ $ORPHAN_COUNT -gt 0 ]; then
    echo -e "${RED}[结论] 检查不通过：Level 2孤立（退出码 3）${NC}"
    exit 3
fi

if [ $PLACEHOLDER_COUNT -gt 0 ]; then
    echo -e "${RED}[结论] 检查不通过：内容为占位符，G1未完成（退出码 5）${NC}"
    exit 5
fi

echo -e "${GREEN}[结论] 检查通过：Level 1骨架完整且有实质内容（退出码 0）${NC}"
exit 0
```

## 使用方法

### 在CI/CD中集成

```yaml
# .github/workflows/analysis-gate.yml
name: 分析门禁检查

on:
  push:
    paths:
      - '系统分析产出/**'

jobs:
  gate-check:
    runs-on: ubuntu-latest
    
    steps:
      - name: 检出代码
        uses: actions/checkout@v3
      
      - name: G1门禁检查
        run: |
          chmod +x ./check-level1-skeleton.sh
          ./check-level1-skeleton.sh "./系统分析产出"
          
      - name: 记录检查点
        if: success()
        run: |
          echo "## 检查点记录" >> checkpoint.md
          echo "- 时间: $(date)" >> checkpoint.md
          echo "- Gate: G1" >> checkpoint.md
          echo "- 脚本退出码: 0" >> checkpoint.md
          git add checkpoint.md
          git commit -m "G1门禁通过"
```

### 在本地执行

```powershell
# Windows
.\check-level1-skeleton.ps1 -RootPath "D:\mywork\techdoc\系统分析产出"

# 检查退出码
if ($LASTEXITCODE -eq 0) {
    Write-Host "门禁通过，可进入G2"
} else {
    Write-Host "门禁失败，退出码: $LASTEXITCODE"
}
```

```bash
# Linux/macOS
./check-level1-skeleton.sh "/path/to/系统分析产出"

# 检查退出码
if [ $? -eq 0 ]; then
    echo "门禁通过，可进入G2"
else
    echo "门禁失败，退出码: $?"
fi
```
